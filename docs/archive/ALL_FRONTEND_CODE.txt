==================== SRC/APP/PERSONALIZE/PAGE.TSX ====================
'use client'

import { Suspense } from 'react'
import { useState, useEffect } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import PrioritySelector from '@/components/personalize/PrioritySelector'
import ScoreBreakdown from '@/components/personalize/ScoreBreakdown'
import StrengthCard from '@/components/personalize/StrengthCard'
import TradeoffCard from '@/components/personalize/TradeoffCard'
import AISummaryCard from '@/components/personalize/AISummaryCard'
import ErrorState from '@/components/personalize/ErrorState'
import LoadingState from '@/components/personalize/LoadingState'
import Button from '@/components/ui/Button'
import apiClient from '@/lib/api'

type PersonalizationStep = 'select' | 'results' | 'summary'

interface PersonalizationResult {
  personalizedScore: number
  factorBreakdown: Record<string, number>
  strongestFactors: string[]
  weakestFactors: string[]
}

function PersonalizeContent() {
  const router = useRouter()
  const searchParams = useSearchParams()
  const neighborhoodId = searchParams.get('neighborhoodId')

  const [step, setStep] = useState<PersonalizationStep>('select')
  const [selectedPriorities, setSelectedPriorities] = useState<string[]>([])
  const [result, setResult] = useState<PersonalizationResult | null>(null)
  const [aiSummary, setAiSummary] = useState<string>('')
  const [isLoading, setIsLoading] = useState(false)
  const [isGeneratingInsight, setIsGeneratingInsight] = useState(false)
  const [error, setError] = useState<string>('')

  useEffect(() => {
    if (!neighborhoodId) {
      setError('No neighborhood selected')
    }
  }, [neighborhoodId])

  const handlePrioritiesSelected = async (priorities: string[]) => {
    setSelectedPriorities(priorities)
    setIsLoading(true)
    setError('')

    try {
      const response = await apiClient.post('/personalize', {
        neighborhood_id: parseInt(neighborhoodId!),
        priorities,
      })

      setResult(response.data)
      setStep('results')

      // Generate AI insight
      await generateAISummary(priorities, response.data)
    } catch (err: any) {
      console.error('Personalize API error:', {
        status: err.response?.status,
        detail: err.response?.data?.detail,
        message: err.message,
      })
      setError(
        err.response?.data?.detail ||
          'Failed to calculate personalized score. Please try again.'
      )
    } finally {
      setIsLoading(false)
    }
  }

  const generateAISummary = async (
    priorities: string[],
    result: PersonalizationResult
  ) => {
    setIsGeneratingInsight(true)

    try {
      const response = await apiClient.post('/summary', {
        neighborhood_id: parseInt(neighborhoodId!),
        priorities,
        personalizedScore: result.personalizedScore,
        factorBreakdown: result.factorBreakdown,
      })

      setAiSummary(response.data.summary)
      setStep('summary')
    } catch (err: any) {
      console.error('Failed to generate AI summary:', err)
      setStep('summary')
      setAiSummary('Unable to generate AI insight at this time.')
    } finally {
      setIsGeneratingInsight(false)
    }
  }

  const handleReset = () => {
    setSelectedPriorities([])
    setResult(null)
    setAiSummary('')
    setError('')
    setStep('select')
  }

  if (!neighborhoodId) {
    return (
      <ErrorState
        message="No neighborhood selected. Please go back and select a neighborhood."
        onRetry={() => router.back()}
      />
    )
  }

  if (error && step === 'select') {
    return <ErrorState message={error} onRetry={() => window.location.reload()} />
  }

  if (isLoading) {
    return <LoadingState />
  }

  return (
    <main className="min-h-screen bg-white">
      <div className="container mx-auto px-4 py-12">
        <div className="max-w-2xl mx-auto">
          {/* Header */}
          <div className="mb-8">
            <h1 className="text-3xl font-bold text-gray-900 mb-2">
              Personalize Your Score
            </h1>
            <p className="text-gray-600">
              Choose what matters most to you, and we&apos;ll personalize the neighborhood
              score based on your priorities.
            </p>
          </div>

          {/* Step: Select Priorities */}
          {step === 'select' && (
            <PrioritySelector
              onPrioritiesSelected={handlePrioritiesSelected}
              isLoading={isLoading}
            />
          )}

          {/* Step: Show Results */}
          {step === 'results' && result && (
            <div className="space-y-8">
              <ScoreBreakdown
                personalizedScore={result.personalizedScore}
                priorities={selectedPriorities}
              />

              <div>
                <h3 className="text-lg font-semibold text-gray-900 mb-3">Strengths</h3>
                <div className="space-y-2">
                  {result.strongestFactors.map((factor) => (
                    <StrengthCard key={factor} factor={factor} />
                  ))}
                </div>
              </div>

              <div>
                <h3 className="text-lg font-semibold text-gray-900 mb-3">
                  Trade-offs to Consider
                </h3>
                <div className="space-y-2">
                  {result.weakestFactors.map((factor) => (
                    <TradeoffCard key={factor} factor={factor} />
                  ))}
                </div>
              </div>
            </div>
          )}

          {/* Step: Show AI Summary */}
          {step === 'summary' && result && (
            <div className="space-y-8">
              <ScoreBreakdown
                personalizedScore={result.personalizedScore}
                priorities={selectedPriorities}
              />

              <div>
                <h3 className="text-lg font-semibold text-gray-900 mb-3">Strengths</h3>
                <div className="space-y-2">
                  {result.strongestFactors.map((factor) => (
                    <StrengthCard key={factor} factor={factor} />
                  ))}
                </div>
              </div>

              <div>
                <h3 className="text-lg font-semibold text-gray-900 mb-3">
                  Trade-offs to Consider
                </h3>
                <div className="space-y-2">
                  {result.weakestFactors.map((factor) => (
                    <TradeoffCard key={factor} factor={factor} />
                  ))}
                </div>
              </div>

              <AISummaryCard
                summary={aiSummary}
                isLoading={isGeneratingInsight}
                error={error}
              />

              <div className="flex gap-3">
                <Button onClick={handleReset} variant="outline" className="flex-1">
                  Adjust Priorities
                </Button>
                <Button
                  onClick={() => router.push('/summary')}
                  className="flex-1"
                >
                  Continue
                </Button>
              </div>
            </div>
          )}
        </div>
      </div>
    </main>
  )
}

export default function Personalize() {
  return (
    <Suspense fallback={<LoadingState />}>
      <PersonalizeContent />
    </Suspense>
  )
}

import clsx from 'clsx'

interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  children: React.ReactNode
  variant?: 'primary' | 'secondary' | 'outline'
  size?: 'sm' | 'md' | 'lg'
}

export default function Button({
  children,
  variant = 'primary',
  size = 'md',
  className,
  ...props
}: ButtonProps) {
  const baseStyles =
    'font-semibold rounded transition-colors focus:outline-none focus:ring-2 focus:ring-offset-2'

  const variants = {
    primary:
      'bg-blue-600 text-white hover:bg-blue-700 focus:ring-blue-500 disabled:bg-gray-400',
    secondary: 'bg-gray-200 text-gray-900 hover:bg-gray-300 focus:ring-gray-500',
    outline:
      'border-2 border-blue-600 text-blue-600 hover:bg-blue-50 focus:ring-blue-500',
  }

  const sizes = {
    sm: 'px-3 py-1 text-sm',
    md: 'px-4 py-2 text-base',
    lg: 'px-6 py-3 text-lg',
  }

  return (
    <button
      className={clsx(baseStyles, variants[variant], sizes[size], className)}
      {...props}
    >
      {children}
    </button>
  )
}
